import { useCallback, useEffect, useState } from 'react'

const CATEGORIES = [
  ['base', 'Les bases'],
  ['proteine', 'Les protéines'],
  ['accompagnement', 'Les accompagnements'],
  ['legume', 'Les légumes'],
  ['sauce', 'Les sauces'],

const PRESETS = [1000, 1500, 2000, 3000]

const ORDER = ['base', 'proteine', 'accompagnement', 'legume', 'sauce']

// Poids d'une portion "parfaite" (en grammes), par ingrédient puis par catégorie
const GRAMS_BY_CAT = { base: 150, proteine: 120, accompagnement: 100, legume: 80, sauce: 25 }
const GRAMS = {
  riz: 150, 'riz-jaune': 150, attieke: 150, 'spaghetti-tomate': 150, 'spaghetti-bolognaise': 150, 'pain-plat': 60,
  'poulet-braise': 150, 'poisson-braise': 150, 'steak-grille': 150, oeuf: 60, jambon: 40, thon: 80,
  frites: 100, 'mais-grille': 150, alloco: 100, avocat: 80, tomate: 60, oignon: 30, piment: 5,
  mayonnaise: 25, ketchup: 20,
}
const gramsOf = (ing) => GRAMS[slug(ing.name)] || GRAMS_BY_CAT[ing.category] || 100

// Place n ingrédients dans le bol sans qu'ils se chevauchent.
// x, y = centre et r = rayon, en % de la largeur du bol.
function layout(n) {
  if (n === 0) return []
  if (n === 1) return [{ x: 50, y: 50, r: 30 }]
  const ring = (m, rho, r, start, cx = 50, cy = 50) =>
    Array.from({ length: m }, (_, i) => {
      const t = ((start + (360 / m) * i) * Math.PI) / 180
      return { x: cx + rho * Math.cos(t), y: cy + rho * Math.sin(t), r }
    })
  if (n === 2) return ring(2, 23, 21, 180)
  if (n === 3) return ring(3, 23, 20, -90)
  if (n === 4) return ring(4, 24, 17.5, -135)
  const m = n - 1
  const rho = 30
  const r = Math.min(rho * Math.sin(Math.PI / m) * 0.95, 17)
  return [{ x: 50, y: 50, r: Math.max(7, rho - r - 3) }, ...ring(m, rho, r, -90)]
}


const SAUCE_COLORS = { mayonnaise: ['#FFF6DA', '#E9DDB4'], ketchup: ['#D8281F', '#A11A13'] }
// Filet de sauce en zigzag, tracé pendant que la bouteille passe au-dessus du plat
function Drizzle({ name, q, k }) {
  const sl = slug(name)
  const [fill, edge] = SAUCE_COLORS[sl] || ['#B4651F', '#7A4210']
  const y0 = 30 + k * 12
  const rows = Math.min(2 + q, 5)
  let d = `M 22 ${y0}`
  for (let i = 0; i < rows; i++) {
    const y = y0 + i * 6
    d += i % 2 === 0 ? ` Q 50 ${y - 4} 78 ${y + 3}` : ` Q 50 ${y + 7} 22 ${y + 3}`
  }
  const w = 1.9 + Math.min(q, 3) * 0.4
  return (
    <g>
      <path d={d} pathLength="1" className="drz" stroke={edge} strokeWidth={w + 0.9} fill="none" strokeLinecap="round" />
      <path d={d} pathLength="1" className="drz" stroke={fill} strokeWidth={w} fill="none" strokeLinecap="round" />
      <g>
        <image href={`/img/ingredients/${sl}.png`} width="26" height="26" x="-13" y="-30" transform="rotate(160)" opacity="0">
          <animate attributeName="opacity" values="0;1;1;0" keyTimes="0;0.08;0.9;1" dur="1.5s" fill="freeze" />
        </image>
        <animateMotion dur="1.4s" path={d} fill="freeze" calcMode="spline" keyTimes="0;1" keySplines="0.4 0 0.6 1" />
      </g>
    </g>
  )
}

export default function Composer() {
  const { profile, refreshProfile } = useAuth()
  const [ingredients, setIngredients] = useState([])
  const [qty, setQty] = useState({})
  const [budget, setBudget] = useState(null)
  const [message, setMessage] = useState(null)
  const [ticket, setTicket] = useState(null)
  const [busy, setBusy] = useState(false)
  const [checkout, setCheckout] = useState(false)

  const load = useCallback(async () => {
    const { data } = await supabase.from('ingredients').select('*').order('name')
    setIngredients(data || [])
  }, [])

  useEffect(() => {
    load()
  }, [load])

  const byId = Object.fromEntries(ingredients.map((i) => [i.id, i]))
  const chosen = Object.entries(qty)
    .map(([id, q]) => ({ ing: byId[id], q }))
    .filter((c) => c.ing && c.q > 0)
  const sorted = [...chosen].sort(
    (a, b) => ORDER.indexOf(a.ing.category) - ORDER.indexOf(b.ing.category) || a.ing.name.localeCompare(b.ing.name)
  )
  const foods = sorted.filter((s) => s.ing.category !== 'sauce')
  const sauces = sorted.filter((s) => s.ing.category === 'sauce')
  const spots = layout(foods.length)
  const grams = chosen.reduce((s, c) => s + gramsOf(c.ing) * c.q, 0)
  const has = (cat) => chosen.some((c) => c.ing.category === cat)
  const tip = !has('base')
    ? 'Ajoute une base (≈ 150 g)'
    : !has('proteine')
      ? 'Ajoute une protéine (≈ 120 g)'
      : !has('legume')
        ? 'Ajoute un légume (≈ 80 g)'
        : 'Assiette équilibrée ✔'
  const total = chosen.reduce((s, c) => s + c.ing.unit_price * c.q, 0)
  const pct = budget ? Math.min(100, Math.round((total / budget) * 100)) : 0

  function add(ing) {
    setMessage(null)
    if (!ing.available || ing.stock_qty <= 0) return
    const current = qty[ing.id] || 0
    if (current + 1 > ing.stock_qty) {
      setMessage({ type: 'error', text: `Il n'en reste plus assez : ${ing.name}.` })
      return
    }
    if (budget && total + ing.unit_price > budget) {
      setMessage({ type: 'error', text: `Budget de ${fcfa(budget)} atteint !` })
      return
    }
    setQty({ ...qty, [ing.id]: current + 1 })
  }

  function sub(ing) {
    setMessage(null)
    const current = qty[ing.id] || 0
    if (current > 0) setQty({ ...qty, [ing.id]: current - 1 })
  }

  function chooseBudget(v) {
    setMessage(null)
    if (v && total > v) {
      setMessage({ type: 'error', text: 'Ton plat dépasse déjà ce budget : retire des ingrédients.' })
      return
    }
    setBudget(v)
  }

  async function topUp(amount) {
    const { error } = await supabase.rpc('recharge', { p_amount: amount })
    if (error) setMessage({ type: 'error', text: error.message })
    else await refreshProfile()
  }

  async function validate(method = 'solde') {
    setBusy(true)
    setMessage(null)
    const items = chosen.map((c) => ({ type: 'ingredient', id: c.ing.id, qty: c.q }))
    const { data, error } = await supabase.rpc('place_order', { p_items: items, p_pickup: today(), p_method: method })
    if (error) setMessage({ type: 'error', text: error.message })
    else {
      setQty({})
      setCheckout(false)
      await refreshProfile()
      load()
      setTicket(await fetchTicket(data))
    }
    setBusy(false)
  }

  return (
    <div>
      <Header />
      {ticket && <Ticket ticket={ticket} onClose={() => setTicket(null)} />}
      <main className="container">
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', gap: 8, flexWrap: 'wrap' }}>
          <Link to="/menu">← Menu</Link>
          <span className="pill pill-mint">Solde : {fcfa(profile?.balance)}</span>
        </div>
        <h2 style={{ marginTop: 10 }}>Compose ton plat</h2>

        <div className="card" style={{ marginBottom: 8 }}>
          <div className="tabs" style={{ paddingBottom: 8 }}>
            <span style={{ alignSelf: 'center', fontWeight: 700 }}>Mon budget</span>
            {PRESETS.map((p) => (
              <button key={p} className={budget === p ? 'on' : ''} onClick={() => chooseBudget(p)}>
                {p.toLocaleString('fr-FR')}
              </button>
            ))}
            <button className={!budget ? 'on' : ''} onClick={() => chooseBudget(null)}>
              Sans limite
            </button>
          </div>
          {budget && (
            <div className="budget">
              <div className={pct >= 100 ? 'full' : ''} style={{ width: `${pct}%` }} />
            </div>
          )}
        </div>

        <div className="bowl">
          <img className="bowl-img" src="/img/bol.png" alt="" onError={(e) => (e.currentTarget.style.display = 'none')} />
          {chosen.length === 0 && (
            <div className="bowl-empty">
              Ton bol est vide.
              <br />
              Touche une assiette !
            </div>
          )}
          {foods.map((s, n) => {
            const z = spots[n]
            const size = 2 * z.r * 0.84 * Math.min(1.39, Math.sqrt(s.q))
            return (
              <div key={s.ing.id} className="in-bowl" style={{ left: `${z.x}%`, top: `${z.y}%`, width: `${size}%`, height: `${size}%` }}>
                {/* la clé change à chaque ajout : le coup de cuillère se rejoue */}
                <div key={s.q} className="scoop">
                  <FoodImg folder="ingredients" name={s.ing.name} category={s.ing.category} size="fill" />
                </div>
                {s.q > 1 && <span className="in-bowl-qty">×{s.q}</span>}
              </div>
            )
          })}
          {sauces.length > 0 && (
            <svg className="drizzle" viewBox="0 0 100 100" aria-hidden="true">
              {sauces.map((s, k) => <Drizzle key={`${s.ing.id}-${s.q}`} name={s.ing.name} q={s.q} k={k} />)}
            </svg>
          )}
        </div>

        <div style={{ textAlign: 'center' }}>
          <div className="display" style={{ fontSize: '2.2rem' }}>
            {fcfa(total)}
            {budget && <span className="muted" style={{ fontSize: '1.2rem' }}> / {fcfa(budget)}</span>}
          </div>
          {chosen.length > 0 && (
            <p className="muted" style={{ margin: '2px 0 10px' }}>
              ≈ <strong>{grams} g</strong> · idéal 400 à 500 g · <strong>{tip}</strong>
            </p>
          )}
          <button className="btn-coral btn-big" disabled={busy || chosen.length === 0} onClick={() => setCheckout(true)}>
            Valider et payer
          </button>
          {message && <p className={message.type === 'error' ? 'error' : 'ok'}>{message.text}</p>}
        </div>

        {CATEGORIES.map(([cat, label]) => {
          const list = ingredients.filter((i) => i.category === cat)
          if (list.length === 0) return null
          return (
            <section key={cat} style={{ marginTop: 22 }}>
              <h3>{label}</h3>
              <div className="plates">
                {list.map((ing) => {
                  const out = !ing.available || ing.stock_qty <= 0
                  const low = !out && ing.stock_qty <= ing.alert_threshold
                  const q = qty[ing.id] || 0
                  return (
                    <div key={ing.id} className="plate-wrap" style={{ opacity: out ? 0.4 : 1 }}>
                      <button className={`plate ${q ? 'on' : ''}`} disabled={out} onClick={() => add(ing)}>
                        <span className="plate-disc">
                          <FoodImg folder="ingredients" name={ing.name} category={ing.category} size={64} />
                        </span>
                        <span className="plate-name">{ing.name}</span>
                        <span className="plate-price">
                          {ing.unit_price} F · ≈{gramsOf(ing)} g
                        </span>
                        {q > 0 && <span className="plate-qty">{q}</span>}
                      </button>
                      <span className="plate-low">{out ? 'Épuisé' : low ? 'Bientôt fini' : ' '}</span>
                      {q > 0 && (
                        <button className="mini" onClick={() => sub(ing)}>
                          − retirer
                        </button>
                      )}
                    </div>
                  )
                })}
              </div>
            </section>
          )
        })}
      </main>
      {checkout && (
        <Checkout total={total} balance={profile?.balance} busy={busy} error={message?.type === 'error' ? message.text : null} onPay={validate} onTopUp={topUp} onClose={() => setCheckout(false)} />
      )}
      <TabBar />
    </div>
  )
}

    return (

        src={`/img/${folder}/${s}.png`}

        onError={() => setFailed(true)}
      />
    )
  }
  return (
    <span role="img" aria-label={name} style={{ fontSize: fill ? 'min(16vw, 74px)' : size * 0.75, lineHeight: 1 }}>
      {EMOJI[s] || CAT[category] || '🍽️'}
    </span>
  )
}

import { useState } from 'react'
import { fcfa } from '../hooks/useOrders'

const METHODS = [
  ['solde', 'Mon solde Cantina', '#ffc244'],
  ['wave', 'Wave', '#1dc8ff'],
  ['orange_money', 'Orange Money', '#ff7900'],
  ['carte', 'Carte bancaire', '#6b7280'],
]
const mark = { solde: 'C', wave: 'W', orange_money: 'O', carte: '💳' }

// Fenêtre de paiement. Wave, Orange Money et carte sont simulés (aucun vrai argent).
export default function Checkout({ total, balance, onPay, onTopUp, onClose, busy, error, noSolde }) {
  const [method, setMethod] = useState(noSolde ? 'wave' : 'solde')
  const methods = noSolde ? METHODS.filter(([k]) => k !== 'solde') : METHODS
  const [phone, setPhone] = useState('')
  const [card, setCard] = useState({ num: '', exp: '', cvc: '' })
  const [waiting, setWaiting] = useState(false)

  const short = method === 'solde' && Number(balance || 0) < total
  const phoneOk = /^\d{8,10}$/.test(phone.replace(/\s/g, ''))
  const cardOk = card.num.replace(/\s/g, '').length >= 12 && card.exp.length >= 4 && card.cvc.length >= 3
  const ready = method === 'solde' ? !short : method === 'carte' ? cardOk : phoneOk

  function pay() {
    if (method === 'solde') return onPay('solde')
    setWaiting(true)
    setTimeout(async () => {
      await onPay(method)
      setWaiting(false)
    }, 1600)
  }

  return (
    <div className="co-overlay" onClick={onClose}>
      <div className="co-sheet" onClick={(e) => e.stopPropagation()}>
        <div className="co-head">
          <h3 style={{ margin: 0 }}>Paiement</h3>
          <button className="btn-light" onClick={onClose} disabled={waiting}>Fermer</button>
        </div>
        <div className="co-total">{fcfa(total)}</div>

        <div className="methods">
          {methods.map(([k, label, color]) => (
            <button key={k} className={`method ${method === k ? 'on' : ''}`} onClick={() => !waiting && setMethod(k)}>
              <span className="method-dot" style={{ background: color }}>{mark[k]}</span>
              <span style={{ flex: 1, textAlign: 'left' }}>
                {label}
                {k === 'solde' && <small className="muted" style={{ display: 'block' }}>Disponible : {fcfa(balance)}</small>}
              </span>
              <span className="method-radio" />
            </button>
          ))}
        </div>

        {(method === 'wave' || method === 'orange_money') && (
          <div style={{ marginTop: 12 }}>
            <input placeholder={`Numéro ${method === 'wave' ? 'Wave' : 'Orange Money'}`} inputMode="numeric" value={phone} onChange={(e) => setPhone(e.target.value)} style={{ width: '100%' }} />
            <small className="muted">Tu recevras une demande de confirmation sur ton téléphone (simulée).</small>
          </div>
        )}
        {method === 'carte' && (
          <div style={{ marginTop: 12, display: 'grid', gap: 8 }}>
            <input placeholder="Numéro de carte" inputMode="numeric" value={card.num} onChange={(e) => setCard({ ...card, num: e.target.value })} />
            <div style={{ display: 'flex', gap: 8 }}>
              <input placeholder="MM/AA" value={card.exp} onChange={(e) => setCard({ ...card, exp: e.target.value })} style={{ flex: 1, minWidth: 0 }} />
              <input placeholder="CVC" inputMode="numeric" value={card.cvc} onChange={(e) => setCard({ ...card, cvc: e.target.value })} style={{ flex: 1, minWidth: 0 }} />
            </div>
          </div>
        )}
        {short && (
          <div className="card" style={{ marginTop: 12 }}>
            <p className="error" style={{ margin: '0 0 8px' }}>Solde insuffisant : il te manque {fcfa(total - Number(balance || 0))}.</p>
            {onTopUp && <button className="btn-coral btn-big" style={{ width: '100%' }} onClick={() => onTopUp(Math.max(1000, Math.ceil((total - Number(balance || 0)) / 500) * 500))}>Recharger {fcfa(Math.max(1000, Math.ceil((total - Number(balance || 0)) / 500) * 500))}</button>}
          </div>
        )}
        {error && <p className="error">{error}</p>}

        <button className="btn-coral btn-big" style={{ width: '100%', marginTop: 14 }} disabled={!ready || busy || waiting} onClick={pay}>
          {waiting ? 'Validation en cours…' : `Payer ${fcfa(total)}`}
        </button>
        <p className="muted" style={{ fontSize: '0.78rem', textAlign: 'center', margin: '8px 0 0' }}>
          Paiement simulé pour la démo : aucun vrai argent n'est débité.
        </p>
      </div>
    </div>
  )
}
